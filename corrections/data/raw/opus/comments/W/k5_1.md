```python
from datetime import date, datetime


def is_overdue(due: date, today: date) -> bool:
    """Return True if `today` is strictly after the `due` date."""
    # datetime is a subclass of date, so isinstance(x, date) alone lets one through.
    # Mixing a datetime with a date makes the comparison raise TypeError, and a time
    # component would make "overdue" depend on the hour, so reject datetimes here.
    for name, value in (("due", due), ("today", today)):
        if not isinstance(value, date) or isinstance(value, datetime):
            raise ValueError(f"{name} must be a datetime.date, got {type(value).__name__}")

    # An invoice isn't overdue on its due date. It becomes overdue the next day.
    return today > due
```

Pass `today` in from the caller, not from `date.today()`, so the function is easy to test and you choose which timezone counts as "today."
