```python
from collections.abc import Iterable
from typing import Any


def dedupe_payments(payments: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return one payment per ref, keeping the one with the lowest id."""
    canonical: dict[Any, dict[str, Any]] = {}
    for payment in payments:
        if "id" not in payment or "ref" not in payment:
            raise ValueError(f"payment missing 'id' or 'ref': {payment!r}")
        ref = payment["ref"]
        current = canonical.get(ref)
        # Upstream sends duplicates out of order, so the first one we see might not be
        # the canonical record. Compare ids instead of keeping whichever came first.
        if current is None or payment["id"] < current["id"]:
            canonical[ref] = payment
    return list(canonical.values())
```

The output is ordered by when each ref first appears in the input, and each entry is the original dict, not a copy. Ids must be comparable to each other, so mixing int and str ids raises a TypeError.
