```python
from typing import Any


def dedupe_payments(payments: list[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return one payment per ref, keeping the one with the lowest id, in input order."""
    best: dict[Any, tuple[int, dict[str, Any]]] = {}
    for index, payment in enumerate(payments):
        if not isinstance(payment, dict) or "id" not in payment or "ref" not in payment:
            raise ValueError(f"payment at index {index} must be a dict with 'id' and 'ref'")
        ref = payment["ref"]
        current = best.get(ref)
        if current is None or payment["id"] < current[1]["id"]:
            best[ref] = (index, payment)
    return [payment for _, payment in sorted(best.values(), key=lambda item: item[0])]
```

The function returns the kept records in the order they appear in the input. If two records share both a ref and the lowest id, it keeps the first one.
