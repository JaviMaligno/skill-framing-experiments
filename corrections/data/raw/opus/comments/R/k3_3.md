```python
from collections.abc import Iterable


def dedupe_payments(payments: Iterable[dict]) -> list[dict]:
    """Return one payment per ref, keeping the record with the lowest id."""
    canonical: dict[object, dict] = {}
    for payment in payments:
        if "id" not in payment or "ref" not in payment:
            raise ValueError(f"payment missing 'id' or 'ref': {payment!r}")
        ref = payment["ref"]
        current = canonical.get(ref)
        # Upstream sends duplicates out of order, and the lowest id is the canonical record.
        if current is None or payment["id"] < current["id"]:
            canonical[ref] = payment
    return list(canonical.values())
```

The output lists refs in the order each one first appears in the input. Each ref keeps its lowest-id record no matter when that record arrives.
