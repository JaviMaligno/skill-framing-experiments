**Request:** load `/reports` one page at a time instead of fetching every report.

## Approach
1. **Backend:** paginate the reports list endpoint, sort it in a fixed order, and remove N+1 queries (one extra database query per row).
2. **Database:** add an index on the sort columns, built with `CONCURRENTLY` so the table isn't locked. This migration needs review before it runs in prod.
3. **Frontend:** read the paginated response, keep the page number in the URL, and add prev/next controls.
4. **Rollout:** pagination only kicks in when `?page=` is in the request. The backend can ship first without breaking the current front end, and you can roll back either side on its own. No feature flag is needed because this doesn't touch billing.

## Changes

**`reports/pagination.py`**
```python
from rest_framework.pagination import PageNumberPagination
from rest_framework.request import Request
from django.db.models import QuerySet


class ReportPagination(PageNumberPagination):
    page_size = 25
    page_size_query_param = "page_size"
    max_page_size = 100

    def paginate_queryset(self, queryset: QuerySet, request: Request, view=None) -> list | None:
        # Opt-in until the frontend ships; remove this guard afterwards.
        if self.page_query_param not in request.query_params:
            return None
        return super().paginate_queryset(queryset, request, view)
```

**`reports/views.py`**

```python
from django.db.models import QuerySet
from rest_framework import generics

from .models import Report
from .pagination import ReportPagination
from .serializers import ReportSerializer


class ReportListView(generics.ListAPIView):
    serializer_class = ReportSerializer
    pagination_class = ReportPagination

    def get_queryset(self) -> QuerySet[Report]:
        return (
            Report.objects
            .select_related("owner")          # whatever FKs the serializer touches
            .order_by("-created_at", "-id")   # stable order; id breaks ties
        )
```

**`frontend/src/pages/ReportsPage.tsx`**
```tsx
import { useEffect, useState } from "react";
import { useSearchParams } from "react-router-dom";

type Report = { id: number; title: string; created_at: string };
type Page<T> = { count: number; next: string | null; previous: string | null; results: T[] };

const PAGE_SIZE = 25;

export default function ReportsPage() {
  const [params, setParams] = useSearchParams();
  const page = Number(params.get("page") ?? "1");
  const [data, setData] = useState<Page<Report> | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    const ctrl = new AbortController();
    setLoading(true);
    fetch(`/api/reports/?page=${page}&page_size=${PAGE_SIZE}`, { signal: ctrl.signal })
      .then((r) => r.json() as Promise<Page<Report>>)
      .then(setData)
      .catch((e) => { if (e.name !== "AbortError") throw e; })
      .finally(() => setLoading(false));
    return () => ctrl.abort();
  }, [page]);

  const totalPages = data ? Math.max(1, Math.ceil(data.count / PAGE_SIZE)) : 1;
  const go = (p: number) => setParams({ page: String(p) });

  return (
    <div>
      {/* existing table, rendering data?.results instead of the full list */}
      <nav aria-label="Pagination">
        <button disabled={!data?.previous || loading} onClick={() => go(page - 1)}>Previous</button>
        <span>Page {page} of {totalPages}</span>
        <button disabled={!data?.next || loading} onClick={() => go(page + 1)}>Next</button>
      </nav>
    </div>
  );
}
```
