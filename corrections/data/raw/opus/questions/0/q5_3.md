Here's a slug helper for invoice titles. It uses Django's `slugify`, caps the length to fit `SlugField`'s default of 50, and has an optional companion that makes the slug unique.

**Approach**
1. Add `core/utils/slugs.py`. I don't know your app layout, so put it wherever your shared utils live.
2. Add `core/tests/test_slugs.py`.
3. No migration is needed, and this doesn't touch billing logic, so it doesn't need a flag.

**Code: `core/utils/slugs.py`**
```python
from __future__ import annotations

import re

from django.db.models import Model, QuerySet
from django.utils.text import slugify

DEFAULT_MAX_LENGTH = 50  # Django SlugField default
FALLBACK_SLUG = "invoice"

# Characters that separate words in invoice titles ("2024/001", "INV.17", "Q3_report").
# slugify() would otherwise drop them and glue the tokens together.
_SEPARATORS = re.compile(r"[/\\._|:+]+")


def invoice_slug(title: str | None, max_length: int = DEFAULT_MAX_LENGTH) -> str:
    """Turn an invoice title into an ASCII URL slug.

    "Factura Nº 2024/001 – Café Núñez" -> "factura-no-2024-001-cafe-nunez"
    """
    if max_length < 1:
        raise ValueError("max_length must be >= 1")

    text = _SEPARATORS.sub(" ", title or "")
    slug = slugify(text)  # NFKD -> ASCII, lowercase, spaces/hyphens collapsed to one "-"

    if len(slug) > max_length:
        cut = slug[:max_length]
        # Prefer cutting at a word boundary when one exists in the kept part.
        if "-" in cut and slug[max_length] != "-":
            cut = cut.rsplit("-", 1)[0]
        slug = cut.strip("-")

    return slug or FALLBACK_SLUG[:max_length]


def unique_invoice_slug(
    title: str | None,
    queryset: QuerySet[Model],
    field: str = "slug",
    max_length: int = DEFAULT_MAX_LENGTH,
    exclude_pk: object | None = None,
) -> str:
    """invoice_slug() plus a -2, -3, ... suffix until it is unique in `queryset`."""
    base = invoice_slug(title, max_length)
    qs = queryset.exclude(pk=exclude_pk) if exclude_pk is not None else queryset

    candidate, n = base, 2
    while qs.filter(**{field: candidate}).exists():
        suffix = f"-{n}"
        candidate = f"{base[: max_length - len(suffix)].rstrip('-')}{suffix}"
        n += 1
    return candidate
```

Example use in a model:
```python
def save(self, *args, **kwargs) -> None:
    if not self.slug:
        self.slug = unique_invoice_slug(self.title, Invoice.objects.all(), exclude_pk=self.pk)
    super().save(*args, **kwargs)
```

The uniqueness check is check-then-write, so two concurrent saves can still pick the same slug. Put `unique=True` on the field so the database catches that case; in practice it's rare.

**Tests: `core/tests/test_slugs.py`**
```python
from django.test import SimpleTestCase

from core.utils.slugs import invoice_slug


class InvoiceSlugTests(SimpleTestCase):
    def test_accents_and_symbols(self) -> None:
        self.assertEqual(
            invoice_slug("Factura Nº 2024/001 – Café Núñez"),
            "factura-no-2024-001-cafe-nunez",
        )

    def test_separators_become_hyphens(self) -> None:
        self.assertEqual(invoice_slug("INV.2024_17"), "inv-2024-17")

    def test_whitespace_and_case(self) -> None:
        self.assertEqual(invoice_slug("  Monthly   SaaS  Fee  "), "monthly-saas-fee")

    def test_empty_falls_back(self) -> None:
        for value in ("", None, "€€€", "///"):
            self.assertEqual(invoice_slug(value), "invoice")

    def test_truncates_on_word_boundary(self) -> None:
        slug = invoice_slug("consulting services for the third quarter", max_length=20)
        self.assertEqual(slug, "consulting-services")
        self.assertLessEqual(len(slug), 20)

    def test_never_ends_with_hyphen(self) -> None:
        slug = invoice_slug("a" * 30 + " " + "b" * 30, max_length=31)
        self.assertFalse(slug.endswith("-"))
```
