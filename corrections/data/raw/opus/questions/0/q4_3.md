**`frontend/src/components/CopyLinkButton.tsx`**
```tsx
import { useEffect, useRef, useState } from "react";

type CopyLinkButtonProps = {
  url: string;
  label?: string;
};

async function copyToClipboard(text: string): Promise<void> {
  if (navigator.clipboard && window.isSecureContext) {
    await navigator.clipboard.writeText(text);
    return;
  }
  // Fallback for non-secure contexts (e.g. plain-http preview envs)
  const textarea = document.createElement("textarea");
  textarea.value = text;
  textarea.setAttribute("readonly", "");
  textarea.style.position = "absolute";
  textarea.style.left = "-9999px";
  document.body.appendChild(textarea);
  textarea.select();
  const ok = document.execCommand("copy");
  document.body.removeChild(textarea);
  if (!ok) throw new Error("Copy failed");
}

export function CopyLinkButton({ url, label = "Copy link" }: CopyLinkButtonProps) {
  const [status, setStatus] = useState<"idle" | "copied" | "error">("idle");
  const timer = useRef<number | undefined>(undefined);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  const handleClick = async () => {
    try {
      await copyToClipboard(url);
      setStatus("copied");
    } catch {
      setStatus("error");
    }
    window.clearTimeout(timer.current);
    timer.current = window.setTimeout(() => setStatus("idle"), 2000);
  };

  const text = status === "copied" ? "Copied!" : status === "error" ? "Copy failed" : label;

  return (
    <button
      type="button"
      onClick={handleClick}
      className="copy-link-button"
      aria-label={`${label}: ${url}`}
    >
      {text}
      <span aria-live="polite" className="sr-only">
        {status === "copied" ? "Link copied to clipboard" : ""}
      </span>
    </button>
  );
}
```

**`frontend/src/components/CopyLinkButton.test.tsx`**
```tsx
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { CopyLinkButton } from "./CopyLinkButton";

test("copies the url and shows confirmation", async () => {
  const user = userEvent.setup();
  const writeText = vi.spyOn(navigator.clipboard, "writeText").mockResolvedValue();
  Object.defineProperty(window, "isSecureContext", { value: true, configurable: true });

  render(<CopyLinkButton url="https://app.example.com/invoices/42" />);
  await user.click(screen.getByRole("button"));

  expect(writeText).toHaveBeenCalledWith("https://app.example.com/invoices/42");
  expect(await screen.findByText("Copied!")).toBeInTheDocument();
});
```
If you use Jest instead of Vitest, change `vi.spyOn` to `jest.spyOn`. `userEvent.setup()` installs its own clipboard stub, which is why the spy goes on `navigator.clipboard` after setup.

## How to check it
1. Run `npm test -- CopyLinkButton`. It should pass.
2. Open an invoice locally and click **Copy link**. The button should say "Copied!" for about 2 seconds.
3. Paste the link into a new tab. It should open the same invoice.
4. Paste it into a logged-out or incognito window. It should go to login first and then to the invoice. This is a quick check that the link doesn't skip auth.
5. Try it on a Railway preview deploy. The clipboard API needs HTTPS there, and the fallback handles plain-http previews.
