I'm reading this as a request for a typed, reusable function that turns an invoice title into a URL-safe slug, such as `"ACME Corp — Q3 Invoice #1042"` → `"acme-corp-q3-invoice-1042"`.

**Assumptions (change any that are wrong):**
- The slug is ASCII only. Accents are transliterated, so `"Café Niño"` becomes `cafe-nino`.
- The slug is capped at 50 characters, which matches Django's default `SlugField` `max_length`. It's cut at a hyphen so no word is split.
- An empty result, such as a title made only of emoji or symbols, falls back to `"invoice"`.
- The helper doesn't check whether the slug is already taken. That needs a database lookup, so I've added a separate optional function for it.
- There are no migrations or billing logic, so no feature flag is needed.

## Approach
1. Add `invoices/utils/slugs.py` with `slugify_invoice_title()`. It's a pure function with no database access.
2. Optionally add `unique_invoice_slug()` in the same file, for use when the slug is saved to a model.
3. Add tests in `invoices/tests/test_slugs.py`.

## Code

`invoices/utils/slugs.py`
```python
from __future__ import annotations

import re
import unicodedata
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from django.db.models import Model, QuerySet

DEFAULT_MAX_LENGTH = 50
FALLBACK_SLUG = "invoice"

_NON_ALNUM = re.compile(r"[^a-z0-9]+")


def slugify_invoice_title(
    title: str | None,
    *,
    max_length: int = DEFAULT_MAX_LENGTH,
    fallback: str = FALLBACK_SLUG,
) -> str:
    """Turn an invoice title into an ASCII, lowercase, hyphen-separated slug.

    >>> slugify_invoice_title("ACME Corp — Q3 Invoice #1042")
    'acme-corp-q3-invoice-1042'
    """
    if max_length < 1:
        raise ValueError("max_length must be >= 1")

    text = unicodedata.normalize("NFKD", title or "")
    text = text.encode("ascii", "ignore").decode("ascii").lower()
    slug = _NON_ALNUM.sub("-", text).strip("-")

    if len(slug) > max_length:
        cut = slug[:max_length]
        # Prefer cutting at a word boundary if one exists in the kept part.
        if slug[max_length] != "-" and "-" in cut:
            cut = cut.rsplit("-", 1)[0]
        slug = cut.strip("-")

    return slug or fallback[:max_length]


def unique_invoice_slug(
    title: str | None,
    queryset: QuerySet[Model],
    *,
    field: str = "slug",
    max_length: int = DEFAULT_MAX_LENGTH,
) -> str:
    """Return a slug not already used in `queryset`, adding -2, -3, ... if needed.

    Pass a queryset that excludes the instance being saved, e.g.
    Invoice.objects.exclude(pk=invoice.pk).
    Still keep a unique constraint on the column; this check is not race-proof.
    """
    base = slugify_invoice_title(title, max_length=max_length)
    candidate = base
    n = 2
    while queryset.filter(**{field: candidate}).exists():
        suffix = f"-{n}"
        candidate = f"{base[: max_length - len(suffix)].rstrip('-')}{suffix}"
        n += 1
    return candidate
```

`invoices/tests/test_slugs.py`
```python
import pytest

from invoices.utils.slugs import slugify_invoice_title


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("ACME Corp — Q3 Invoice #1042", "acme-corp-q3-invoice-1042"),
        ("  Café Niño  ", "cafe-nino"),
        ("Invoice / 2026 / 10", "invoice-2026-10"),
        ("---weird___title---", "weird-title"),
        ("", "invoice"),
        (None, "invoice"),
        ("💸💸💸", "invoice"),
    ],
)
def test_slugify_invoice_title(title: str | None, expected: str) -> None:
    assert slugify_invoice_title(title) == expected


def test_truncates_on_word_boundary() -> None:
    slug = slugify_invoice_title("alpha beta gamma delta", max_length=13)
    assert slug == "alpha-beta"
    assert len(slug) <= 13


def test_hard_cut_when_no_boundary() -> None:
    assert slugify_invoice_title("a" * 80, max_length=10) == "a" * 10
```


**Notes:**
- I didn't use Django's `django.utils.text.slugify` because it keeps underscores and has no length limit or fallback.
- If you want non-Latin titles (Cyrillic, CJK) kept rather than removed, say so. That needs `allow_unicode`-style handling, and the URLs will look different.
