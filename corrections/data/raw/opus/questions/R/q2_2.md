This is the field I'd use for whichever option you pick:

```python
from decimal import Decimal, ROUND_HALF_UP
from rest_framework import serializers

class EuroFromCentsField(serializers.Field):
    """Exposes an integer-cents model field as a euro decimal string."""

    def to_representation(self, value: int) -> str:
        return str((Decimal(value) / 100).quantize(Decimal("0.01"), ROUND_HALF_UP))

    def to_internal_value(self, data: str | int | float) -> int:
        try:
            amount = Decimal(str(data))
        except Exception:
            raise serializers.ValidationError("Invalid price.")
        if amount.as_tuple().exponent < -2:
            raise serializers.ValidationError("Price may have at most 2 decimal places.")
        return int(amount * 100)
```
