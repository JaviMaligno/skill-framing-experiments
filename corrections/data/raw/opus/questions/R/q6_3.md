**What you're asking for:** make `/reports` load one page of reports at a time instead of all of them.

**What I'm assuming** (none of it is hard to undo):
- The React `/reports` page gets its data from a DRF endpoint, `GET /api/reports/`, backed by a `Report` model with a `created_at` field.
- It shows 25 per page, newest first, with page-number pagination (Prev/Next and a total count).
- The page number lives in the URL (`?page=3`), so refreshing keeps your place and pages can be linked.

If the page is a server-rendered Django template instead, the same idea works with `ListView` and `paginate_by = 25`. Tell me and I'll rewrite it that way.

## Approach
1. **Backend:** add pagination to this one endpoint only, so other endpoints keep their current response format. Also stop the related-object lookups that run once per report (an N+1 query), which is often half the slowness.
2. **Migration:** add an index that matches the sort order. It goes in a separate migration so it can be reviewed before it runs in production.
3. **Frontend:** read the paginated response and add page controls.
4. **Check other consumers:** the endpoint's response changes from a plain list to `{count, next, previous, results}`, so anything else that calls it will break until it's updated.

## 1. Backend

`reports/pagination.py`
```python
from rest_framework.pagination import PageNumberPagination


class ReportPagination(PageNumberPagination):
    page_size: int = 25
    page_size_query_param: str = "page_size"
    max_page_size: int = 100
```

`reports/views.py`
```python
from django.db.models import QuerySet
from rest_framework import generics

from .models import Report
from .pagination import ReportPagination
from .serializers import ReportSerializer


class ReportListView(generics.ListAPIView):
    serializer_class = ReportSerializer
    pagination_class = ReportPagination

    def get_queryset(self) -> QuerySet[Report]:
        return (
            Report.objects
            .select_related("created_by")  # replace with whatever FKs the serializer reads
            .order_by("-created_at", "-id")  # ordering must be stable or pages overlap
        )
```

## 2. Index migration (review before it runs in production)

`reports/models.py`, inside `Report`:
```python
class Meta:
    indexes = [
        models.Index(fields=["-created_at", "-id"], name="report_created_id_idx"),
    ]
```

Run `python manage.py makemigrations reports`. Then edit the generated migration so it builds the index without locking the table:
```python
from django.contrib.postgres.operations import AddIndexConcurrently
from django.db import migrations, models


class Migration(migrations.Migration):
    atomic = False

    dependencies = [("reports", "XXXX_previous")]

    operations = [
        AddIndexConcurrently(
            model_name="report",
            index=models.Index(fields=["-created_at", "-id"], name="report_created_id_idx"),
        ),
    ]
```
To undo it, migrate back to the previous migration; that drops the index.

## 3. Frontend

`src/pages/Reports.tsx`
```tsx
import { useEffect, useState } from "react";
import { useSearchParams } from "react-router-dom";

type Report = { id: number; title: string; created_at: string };
type Page<T> = { count: number; next: string | null; previous: string | null; results: T[] };

const PAGE_SIZE = 25;

export default function Reports() {
  const [params, setParams] = useSearchParams();
  const page = Math.max(1, Number(params.get("page")) || 1);
  const [data, setData] = useState<Page<Report> | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const ctrl = new AbortController();
    setLoading(true);
    fetch(`/api/reports/?page=${page}&page_size=${PAGE_SIZE}`, { signal: ctrl.signal })
      .then((r) => r.json())
      .then(setData)
      .catch((e) => { if (e.name !== "AbortError") console.error(e); })
      .finally(() => setLoading(false));
    return () => ctrl.abort();
  }, [page]);

  const totalPages = data ? Math.max(1, Math.ceil(data.count / PAGE_SIZE)) : 1;
  const go = (p: number) => setParams({ page: String(p) });

  return (
    <div>
      {loading && !data ? <p>Loading…</p> : (
        <ul>{data?.results.map((r) => <li key={r.id}>{r.title}</li>)}</ul>
      )}
      <nav>
        <button disabled={!data?.previous} onClick={() => go(page - 1)}>Prev</button>
        <span> Page {page} of {totalPages} </span>
        <button disabled={!data?.next} onClick={() => go(page + 1)}>Next</button>
      </nav>
    </div>
  );
}
```
Keep your existing row markup in place of the `<li>`.

## How to check it works
- **Test:**
  ```python
  def test_reports_paginated(client, report_factory) -> None:
      report_factory.create_batch(30)
      r = client.get("/api/reports/?page=2")
      assert r.status_code == 200
      assert r.json()["count"] == 30
      assert len(r.json()["results"]) == 5
  ```
- **Query count:** use `django_assert_num_queries` to confirm one page takes about 2 queries (one for the count, one for the page), not 1 + N.
- **Index:** in `psql`, run `EXPLAIN ANALYZE SELECT ... ORDER BY created_at DESC, id DESC LIMIT 25;`. You should see an index scan on `report_created_id_idx`, not a sort over the whole table.
- **Manually:** open `/reports?page=2`, refresh, and use browser back/forward. The page should stay put, and the network tab should show a single small response.

If the table grows past a few hundred thousand rows, `COUNT(*)` on every request becomes the slow part. At that point switch to `CursorPagination` and drop the "of N" display.
