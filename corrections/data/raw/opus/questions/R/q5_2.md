**Request:** a helper that turns an invoice title into a URL slug.

**Assumptions:**
- It's a pure function in a shared utils module, `billing/utils/slugs.py`. It isn't wired into any model yet.
- Output is ASCII only, so `"Café"` becomes `"cafe"`.
- Maximum length is 50 characters, to match Django's default `SlugField`.
- Titles that produce an empty slug (e.g. `"€€€"`) fall back to `"invoice"`.
- The function doesn't make slugs unique. I've included a separate helper for that.

Nothing here touches billing logic or the database, so it doesn't need a feature flag or a migration.

## Approach
1. Add `billing/utils/slugs.py` with `invoice_slug()` and an optional `unique_invoice_slug()`.
2. Add tests in `billing/tests/test_slugs.py`.

## Code

```python
# billing/utils/slugs.py
from __future__ import annotations

from collections.abc import Callable

from django.utils.text import slugify

DEFAULT_MAX_LENGTH = 50
FALLBACK_SLUG = "invoice"


def invoice_slug(title: str | None, max_length: int = DEFAULT_MAX_LENGTH) -> str:
    """Turn an invoice title into an ASCII URL slug.

    "Factura Nº 2024/017 – Café Olé" -> "factura-no-2024017-cafe-ole"
    """
    if max_length < 1:
        raise ValueError("max_length must be >= 1")

    slug = slugify(title or "", allow_unicode=False)

    if len(slug) > max_length:
        cut = slug[:max_length]
        # Cut at the last hyphen so no word is split, unless that leaves almost nothing.
        head, sep, _ = cut.rpartition("-")
        slug = head if sep and len(head) >= max_length // 2 else cut
        slug = slug.strip("-")

    return slug or FALLBACK_SLUG[:max_length]


def unique_invoice_slug(
    title: str | None,
    exists: Callable[[str], bool],
    max_length: int = DEFAULT_MAX_LENGTH,
) -> str:
    """Return a slug that isn't taken yet, adding -2, -3, ... when needed.

    `exists` checks whether a slug is already used, e.g.
    lambda s: Invoice.objects.filter(slug=s).exists()
    """
    base = invoice_slug(title, max_length)
    if not exists(base):
        return base

    n = 2
    while True:
        suffix = f"-{n}"
        candidate = f"{base[: max_length - len(suffix)].rstrip('-')}{suffix}"
        if not exists(candidate):
            return candidate
        n += 1
```

`unique_invoice_slug` can still race if two invoices are saved at the same moment. If you put this on a model, give the slug column a unique constraint and retry on `IntegrityError`.

## Tests

```python
# billing/tests/test_slugs.py
from django.test import SimpleTestCase

from billing.utils.slugs import invoice_slug, unique_invoice_slug


class InvoiceSlugTests(SimpleTestCase):
    def test_basic(self) -> None:
        self.assertEqual(invoice_slug("March Invoice 2026"), "march-invoice-2026")

    def test_accents_and_symbols(self) -> None:
        self.assertEqual(
            invoice_slug("Factura Nº 2024/017 – Café Olé"),
            "factura-no-2024017-cafe-ole",
        )

    def test_empty_falls_back(self) -> None:
        for value in ("", None, "€€€", "   "):
            self.assertEqual(invoice_slug(value), "invoice")

    def test_truncates_on_word_boundary(self) -> None:
        slug = invoice_slug("consulting services for the quarterly review " * 3)
        self.assertLessEqual(len(slug), 50)
        self.assertFalse(slug.endswith("-"))
        self.assertTrue(slug.endswith(("review", "the", "quarterly", "for", "services", "consulting")))

    def test_unique_adds_suffix(self) -> None:
        taken = {"acme-invoice", "acme-invoice-2"}
        self.assertEqual(unique_invoice_slug("ACME Invoice", taken.__contains__), "acme-invoice-3")

    def test_unique_respects_max_length(self) -> None:
        taken = {"a" * 10}
        self.assertEqual(unique_invoice_slug("a" * 30, taken.__contains__, max_length=10), "aaaaaaaa-2")
```
