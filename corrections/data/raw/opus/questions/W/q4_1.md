## Change

**`frontend/src/components/CopyLinkButton.tsx`** (new)
```tsx
import { useEffect, useRef, useState } from "react";

type CopyLinkButtonProps = {
  /** Absolute or relative URL. Relative paths are resolved against the current origin. */
  url: string;
  label?: string;
  className?: string;
};

async function copyText(text: string): Promise<void> {
  if (navigator.clipboard && window.isSecureContext) {
    await navigator.clipboard.writeText(text);
    return;
  }
  // Fallback for non-secure contexts / older browsers
  const textarea = document.createElement("textarea");
  textarea.value = text;
  textarea.setAttribute("readonly", "");
  textarea.style.position = "fixed";
  textarea.style.opacity = "0";
  document.body.appendChild(textarea);
  textarea.select();
  try {
    if (!document.execCommand("copy")) throw new Error("copy command failed");
  } finally {
    document.body.removeChild(textarea);
  }
}

export function CopyLinkButton({ url, label = "Copy link", className }: CopyLinkButtonProps) {
  const [status, setStatus] = useState<"idle" | "copied" | "error">("idle");
  const timer = useRef<number | undefined>(undefined);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  const handleClick = async () => {
    const absoluteUrl = new URL(url, window.location.origin).toString();
    try {
      await copyText(absoluteUrl);
      setStatus("copied");
    } catch {
      setStatus("error");
    }
    window.clearTimeout(timer.current);
    timer.current = window.setTimeout(() => setStatus("idle"), 2000);
  };

  const text = status === "copied" ? "Copied!" : status === "error" ? "Copy failed" : label;

  return (
    <button
      type="button"
      onClick={handleClick}
      className={className}
      aria-label={`${label} to invoice`}
      title={label}
      data-testid="copy-invoice-link"
    >
      {text}
      <span role="status" aria-live="polite" className="sr-only">
        {status === "copied" ? "Link copied to clipboard" : ""}
      </span>
    </button>
  );
}
```

## Assumptions (adjust if wrong)
- **Paths:** the page is `frontend/src/pages/InvoiceDetail.tsx` and uses TSX. If it's plain JSX, remove the types.
- **Link format:** the link is the invoice's front-end route, `/invoices/<id>`, built from the ID rather than `window.location.href`. That keeps query strings and tab state out of the link. If you'd rather share the invoice number (`/invoices/INV-0042`) or a Django URL, change the `url` prop.
- **Styling:** `btn btn-sm btn-secondary` stands in for your button classes. If you have an icon set, a clipboard icon with the text as a tooltip is a natural swap.
- **Who can open the link:** it goes to the normal authenticated page, so recipients still need access to the invoice. It isn't a public share link. If you wanted a link that customers or people outside the team could open, tell me. That would need backend work (signed tokens) and would go behind a flag.
