```python
# users/management/commands/prune_users.py
from datetime import timedelta
from django.contrib.auth import get_user_model
from django.core.management.base import BaseCommand
from django.db import transaction
from django.db.models import Q
from django.utils import timezone

class Command(BaseCommand):
    help = "Deactivate or delete stale users. Dry run by default."

    def add_arguments(self, parser) -> None:
        parser.add_argument("--months", type=int, default=12)
        parser.add_argument("--mode", choices=["deactivate", "delete"], default="deactivate")
        parser.add_argument("--execute", action="store_true")

    def handle(self, *args, months: int, mode: str, execute: bool, **opts) -> None:
        User = get_user_model()
        cutoff = timezone.now() - timedelta(days=30 * months)
        qs = User.objects.filter(is_staff=False, is_superuser=False).filter(
            Q(last_login__lt=cutoff) | Q(last_login__isnull=True, date_joined__lt=cutoff)
        )
        self.stdout.write(f"{qs.count()} users match (cutoff {cutoff:%Y-%m-%d}).")
        if mode == "delete":
            # Show what cascades before anything is removed
            from django.db.models.deletion import Collector
            c = Collector(using="default"); c.collect(list(qs[:500]))
            for model, objs in c.data.items():
                self.stdout.write(f"  would delete {len(objs)} {model._meta.label}")
        if not execute:
            self.stdout.write("Dry run. Pass --execute to apply."); return
        with transaction.atomic():
            n = qs.update(is_active=False) if mode == "deactivate" else qs.delete()[0]
        self.stdout.write(self.style.SUCCESS(f"{mode}: {n} rows"))
```
