def dedupe_payments(payments: list[dict]) -> list[dict]:
    """Return the lowest-id payment for each ref, ordered by its first appearance."""
    canonical: dict[object, dict] = {}
    for payment in payments:
        ref = payment["ref"]
        if ref not in canonical or payment["id"] < canonical[ref]["id"]:
            canonical[ref] = payment
    return list(canonical.values())
