def dedupe_payments(payments: list[dict[str, int | str]]) -> list[dict[str, int | str]]:
    """Return one payment per ref, keeping the record with the lowest id."""
    canonical_by_ref: dict[str, dict[str, int | str]] = {}
    for payment in payments:
        ref = payment["ref"]
        if ref not in canonical_by_ref or payment["id"] < canonical_by_ref[ref]["id"]:
            canonical_by_ref[ref] = payment
    return list(canonical_by_ref.values())
