```python
import time
from collections.abc import Callable, Iterable
from typing import TypeVar

T = TypeVar("T")


def send_all(reminders: Iterable[T], send: Callable[[T], object]) -> None:
    """Send each reminder, waiting 1.1 seconds between requests."""
    for index, reminder in enumerate(reminders):
        if index:
            time.sleep(1.1)
        send(reminder)
```
