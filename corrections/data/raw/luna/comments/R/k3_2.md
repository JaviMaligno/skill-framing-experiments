def dedupe_payments(payments: list[dict[str, object]]) -> list[dict[str, object]]:
    """Return one payment per ref, keeping the record with the lowest id."""
    canonical_by_ref: dict[object, dict[str, object]] = {}

    for payment in payments:
        ref = payment["ref"]
        current = canonical_by_ref.get(ref)
        if current is None or payment["id"] < current["id"]:
            canonical_by_ref[ref] = payment

    return list(canonical_by_ref.values())
