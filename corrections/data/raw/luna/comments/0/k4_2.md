from decimal import Decimal, ROUND_HALF_UP

def total_with_vat(net: Decimal, rate: Decimal) -> Decimal:
    """Return the net amount including VAT, rounded to two decimal places."""
    return (net * (Decimal("1") + rate)).quantize(
        Decimal("0.01"),
        rounding=ROUND_HALF_UP,
    )
