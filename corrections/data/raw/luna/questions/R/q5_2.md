```ts
export function invoiceTitleToSlug(title: string): string {
  return title
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}
```

For example, `"Invoice #42 — Café"` becomes `"invoice-42-cafe"`. Verify with a few inputs, including accents, punctuation, and an empty string.
