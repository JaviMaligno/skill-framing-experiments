```python
import re
import unicodedata


def invoice_title_to_slug(title: str) -> str:
    """Convert an invoice title into a lowercase, hyphen-separated URL slug."""
    normalized = unicodedata.normalize("NFKD", title)
    ascii_text = normalized.encode("ascii", "ignore").decode("ascii")
    slug = re.sub(r"[^a-zA-Z0-9]+", "-", ascii_text.lower())
    return slug.strip("-")
```
