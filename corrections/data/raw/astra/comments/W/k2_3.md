from collections.abc import Callable, Iterable
from time import sleep


def send_all[T](reminders: Iterable[T], send: Callable[[T], object]) -> None:
    """Return None after sending every reminder with rate-limit spacing."""
    for index, reminder in enumerate(reminders):
        if index:
            # The provider silently drops excess requests; keep a 0.1 s margin.
            sleep(1.1)
        send(reminder)
