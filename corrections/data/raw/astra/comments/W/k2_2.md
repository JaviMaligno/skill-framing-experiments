from collections.abc import Callable, Iterable
from time import sleep


def send_all[T](reminders: Iterable[T], send: Callable[[T], object]) -> None:
    """Return None after sending each reminder with rate-limit spacing."""
    first = True
    for reminder in reminders:
        if not first:
            # The extra 0.1 s guards against the provider silently dropping requests.
            sleep(1.1)
        send(reminder)
        first = False
