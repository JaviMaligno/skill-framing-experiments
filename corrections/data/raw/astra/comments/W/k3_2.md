```python
from collections.abc import Iterable
from typing import Any


def dedupe_payments(payments: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return the lowest-id payment per ref, in first-seen ref order."""
    canonical: dict[Any, dict[str, Any]] = {}

    for payment in payments:
        if not isinstance(payment, dict) or not {"id", "ref"} <= payment.keys():
            raise ValueError("Each payment must be a dict with 'id' and 'ref'.")

        ref = payment["ref"]
        try:
            if ref not in canonical or payment["id"] < canonical[ref]["id"]:
                canonical[ref] = payment
        except TypeError as exc:
            raise ValueError(
                "Refs must be hashable and IDs for the same ref must be comparable."
            ) from exc

    return list(canonical.values())
```

Equal IDs keep the first encountered record; input dictionaries are not modified.
