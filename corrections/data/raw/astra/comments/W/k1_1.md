from datetime import date, timedelta


def billing_period_end(api_end: date) -> date:
    """Return the last billed day for an exclusive API period end."""
    if type(api_end) is not date or api_end == date.min:
        raise ValueError("api_end must be a date later than date.min")

    # The bank's end date excludes that day; invoices show an inclusive end.
    return api_end - timedelta(days=1)
