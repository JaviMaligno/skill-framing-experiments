from datetime import date, timedelta


def billing_period_end(api_end: date) -> date:
    """Return the last billed day for an exclusive API period end."""
    if not isinstance(api_end, date):
        raise ValueError("api_end must be a date")
    if api_end == date.min:
        raise ValueError("api_end must be later than date.min")

    # The bank's exclusive boundary is one day after the invoice's inclusive end.
    return api_end - timedelta(days=1)
