```python
from collections.abc import Iterable
from typing import Any


def dedupe_payments(payments: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return the lowest-id payment per ref, in first-seen ref order."""
    canonical: dict[Any, dict[str, Any]] = {}

    try:
        for payment in payments:
            if not isinstance(payment, dict):
                raise ValueError("Each payment must be a dict.")
            ref, payment_id = payment["ref"], payment["id"]
            if ref not in canonical or payment_id < canonical[ref]["id"]:
                canonical[ref] = payment
    except (KeyError, TypeError) as exc:
        raise ValueError(
            "Payments need 'id' and 'ref', hashable refs, and comparable ids."
        ) from exc

    return list(canonical.values())
```

Equal IDs keep the first record encountered; input records are not modified.
