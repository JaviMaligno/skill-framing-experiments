```python
from collections.abc import Callable, Iterable
from time import sleep


def send_all[T](reminders: Iterable[T], send: Callable[[T], object]) -> None:
    """Return None after sending all reminders at least 1.1 seconds apart."""
    for index, reminder in enumerate(reminders):
        if index:
            sleep(1.1)
        send(reminder)
```
