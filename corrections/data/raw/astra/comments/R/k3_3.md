```python
from collections.abc import Iterable
from typing import Any


def dedupe_payments(payments: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return the payment with the lowest id for each ref."""
    canonical: dict[str, dict[str, Any]] = {}

    for payment in payments:
        if not isinstance(payment, dict):
            raise ValueError("Each payment must be a dict")
        if not isinstance(payment.get("ref"), str):
            raise ValueError("Each payment must have a string ref")
        if type(payment.get("id")) is not int:
            raise ValueError("Each payment must have an integer id")

        ref = payment["ref"]
        if ref not in canonical or payment["id"] < canonical[ref]["id"]:
            canonical[ref] = payment

    return list(canonical.values())
```

Assumes integer IDs and string refs. Results follow first-seen ref order; equal IDs keep the first record.
