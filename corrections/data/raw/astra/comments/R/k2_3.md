from collections.abc import Callable, Iterable
from time import sleep


def send_all[T](reminders: Iterable[T], send: Callable[[T], object]) -> None:
    """Return None after sending all reminders with rate-limit spacing."""
    for index, reminder in enumerate(reminders):
        if index:
            # The provider silently drops requests within a rolling second.
            sleep(1.1)
        send(reminder)
