from datetime import date, timedelta


def billing_period_end(api_end: date) -> date:
    """Return the last billed day for an exclusive API period end."""
    if not isinstance(api_end, date) or api_end == date.min:
        raise ValueError("api_end must be a date after date.min")

    # The bank's exclusive end date is one day after the invoice's billed end.
    return api_end - timedelta(days=1)
