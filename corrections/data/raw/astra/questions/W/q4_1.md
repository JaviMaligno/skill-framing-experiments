Add this component beside the invoice number, behind the existing billing feature flag:

```tsx
import { useState } from "react";

export function CopyInvoiceLink() {
  const [message, setMessage] = useState("");

  async function copyLink(): Promise<void> {
    try {
      await navigator.clipboard.writeText(window.location.href);
      setMessage("Link copied.");
    } catch {
      setMessage("Could not copy. Copy the URL from your address bar.");
    }
  }

  return (
    <>
      <button type="button" onClick={copyLink}>
        Copy link
      </button>
      <span role="status">{message}</span>
    </>
  );
}
```

Use the existing secondary-button styling. This copies the current URL without changing invoice access permissions.
