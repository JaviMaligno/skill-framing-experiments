**Proposed change:** add this React component beside the invoice number, behind the existing billing feature-flag system.

```tsx
import { useState } from "react";

export function CopyInvoiceLink({ url }: { url: string }) {
  const [message, setMessage] = useState("");

  async function copyLink(): Promise<void> {
    try {
      await navigator.clipboard.writeText(url);
      setMessage("Link copied.");
    } catch {
      setMessage("Could not copy. Copy the URL from your address bar.");
    }
  }

  return (
    <>
      <button type="button" onClick={copyLink}>
        Copy link
      </button>
      <span role="status">{message}</span>
    </>
  );
}
```
