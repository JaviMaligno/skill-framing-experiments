Assuming React and that “Copy link” means the current page URL, add this component beside the invoice number, behind the existing billing feature flag:

```tsx
import { useState } from "react";

export function CopyInvoiceLink() {
  const [status, setStatus] = useState("");

  async function copyLink(): Promise<void> {
    try {
      await navigator.clipboard.writeText(window.location.href);
      setStatus("Link copied");
    } catch {
      setStatus("Couldn't copy link. Copy the URL from your address bar.");
    }
  }

  return (
    <>
      <button type="button" onClick={copyLink}>
        Copy link
      </button>
      <span role="status">{status}</span>
    </>
  );
}
```
