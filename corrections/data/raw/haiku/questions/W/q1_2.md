- "Old" means `last_login` is more than 365 days ago (or never set and `date_joined` is over 365 days ago), and the user is not staff or superuser.
- Deactivating is safer than deleting. `is_active=False` hides users from admin and can be undone. Deleting cascades through related rows and may fail on `PROTECT` foreign keys.

**Dry-run management command** (run it from your repo, then review the count before adding `--apply`):

```python
# <app>/management/commands/prune_stale_users.py
from datetime import timedelta
from typing import Any

from django.contrib.auth import get_user_model
from django.core.management.base import BaseCommand
from django.db.models import Q
from django.utils import timezone

User = get_user_model()


class Command(BaseCommand):
    help = "Deactivate (default) or delete stale users. Dry run unless --apply is passed."

    def add_arguments(self, parser: Any) -> None:
        parser.add_argument("--days", type=int, default=365)
        parser.add_argument("--delete", action="store_true", help="Hard delete instead of deactivating")
        parser.add_argument("--apply", action="store_true")

    def handle(self, *args: Any, **opts: Any) -> None:
        cutoff = timezone.now() - timedelta(days=opts["days"])
        qs = (
            User.objects.filter(is_staff=False, is_superuser=False)
            .filter(Q(last_login__lt=cutoff) | Q(last_login__isnull=True, date_joined__lt=cutoff))
        )
        count = qs.count()
        self.stdout.write(f"{count} users match (cutoff {cutoff:%Y-%m-%d})")

        if not opts["apply"]:
            self.stdout.write("Dry run. Re-run with --apply to change anything.")
            return

        if opts["delete"]:
            deleted, per_model = qs.delete()
            self.stdout.write(f"Deleted {deleted} rows: {per_model}")
        else:
            updated = qs.filter(is_active=True).update(is_active=False)
            self.stdout.write(f"Deactivated {updated} users")
```
