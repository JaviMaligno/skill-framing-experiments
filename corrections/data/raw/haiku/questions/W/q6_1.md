**Approach**
1. `reports/views.py`: switch the list to a paginated `ListView`, with a stable ordering and `select_related` to avoid per-row queries.
2. `reports/templates/reports/report_list.html`: render the page controls.
3. `reports/tests.py`: add tests for page size and page navigation.
4. No migration is needed for this step.

**1. View**

```python
# reports/views.py
from django.views.generic import ListView

from .models import Report


class ReportListView(ListView):
    template_name = "reports/report_list.html"
    context_object_name = "reports"
    paginate_by = 25

    def get_queryset(self) -> "QuerySet[Report]":
        return (
            Report.objects
            .select_related("owner")  # add every FK the template touches
            .order_by("-created_at", "pk")  # pk breaks ties so pages don't overlap
        )
```

```django
    <a href="?page={{ page_obj.next_page_number }}">Next</a>
  {% endif %}

```

Out-of-range page numbers return 404 from `ListView`, which is standard behaviour.

**3. Tests**

```python
# reports/tests.py
from django.test import TestCase
from django.urls import reverse

from .models import Report


class ReportListPaginationTests(TestCase):
    @classmethod
    def setUpTestData(cls):
        Report.objects.bulk_create([Report(title=f"Report {i}") for i in range(60)])

    def test_first_page_has_25_rows_and_pagination(self):
        resp = self.client.get(reverse("reports:list"))
        self.assertEqual(len(resp.context["reports"]), 25)
        self.assertTrue(resp.context["is_paginated"])
        self.assertEqual(resp.context["paginator"].num_pages, 3)

    def test_last_page_has_remainder(self):
        resp = self.client.get(reverse("reports:list"), {"page": 3})
        self.assertEqual(len(resp.context["reports"]), 10)

    def test_query_count_is_constant_per_page(self):
        with self.assertNumQueries(3):  # count + page + one per select_related? adjust to actual
            self.client.get(reverse("reports:list"))
```
