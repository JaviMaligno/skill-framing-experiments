**Approach**
1. Backend: paginate in the query with `Paginator`, fetch only the columns the list needs, and avoid N+1 queries on `owner`.
2. Frontend: track the page in state, request `?page=`, and add Prev/Next.

**1. Backend: `reports/views.py`**

```python
from __future__ import annotations

from django.core.paginator import EmptyPage, Paginator
from django.http import HttpRequest, JsonResponse
from django.views.decorators.http import require_GET

from reports.models import Report

DEFAULT_PAGE_SIZE = 25
MAX_PAGE_SIZE = 100


@require_GET
def report_list(request: HttpRequest) -> JsonResponse:
    try:
        page_number = int(request.GET.get("page", 1))
        page_size = int(request.GET.get("page_size", DEFAULT_PAGE_SIZE))
    except ValueError:
        return JsonResponse({"error": "page and page_size must be integers"}, status=400)

    page_size = min(max(page_size, 1), MAX_PAGE_SIZE)

    # Stable ordering is required for paging; the tiebreaker on id prevents duplicates across pages.
    qs = (
        Report.objects.select_related("owner")
        .only("id", "title", "created_at", "owner__id", "owner__username")
        .order_by("-created_at", "-id")
    )

    paginator = Paginator(qs, page_size)
    try:
        page = paginator.page(page_number)
    except EmptyPage:
        return JsonResponse({"error": "page out of range"}, status=404)

    return JsonResponse({
        "results": [
            {
                "id": r.id,
                "title": r.title,
                "created_at": r.created_at.isoformat(),
                "owner": r.owner.username,
            }
            for r in page.object_list
        ],
        "page": page.number,
        "num_pages": paginator.num_pages,
        "count": paginator.count,
        "has_next": page.has_next(),
        "has_previous": page.has_previous(),
    })
```

**2. Frontend: the `/reports` component**

```tsx
import { useEffect, useState } from "react";

type Report = { id: number; title: string; created_at: string; owner: string };
type ReportPage = {
  results: Report[];
  page: number;
  num_pages: number;
  count: number;
  has_next: boolean;
  has_previous: boolean;
};

export function ReportsPage() {
  const [page, setPage] = useState(1);
  const [data, setData] = useState<ReportPage | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const controller = new AbortController();
    setError(null);
    fetch(`/api/reports/?page=${page}&page_size=25`, {
      signal: controller.signal,
      credentials: "same-origin",
    })
      .then((r) => {
        if (!r.ok) throw new Error(`HTTP ${r.status}`);
        return r.json() as Promise<ReportPage>;
      })
      .then(setData)
      .catch((e: unknown) => {
        if (e instanceof DOMException && e.name === "AbortError") return;
        setError(String(e));
      });
    return () => controller.abort();
  }, [page]);

  if (error) return <p>Failed to load reports: {error}</p>;
  if (!data) return <p>Loading…</p>;

  return (
    <div>
      <ul>
        {data.results.map((r) => (
          <li key={r.id}>
            {r.title} — {r.owner} — {new Date(r.created_at).toLocaleDateString()}
          </li>
        ))}
      </ul>
      <nav>
        <button onClick={() => setPage((p) => p - 1)} disabled={!data.has_previous}>
          Previous
        </button>
        <span>
          Page {data.page} of {data.num_pages} ({data.count} reports)
        </span>
        <button onClick={() => setPage((p) => p + 1)} disabled={!data.has_next}>
          Next
        </button>
      </nav>
    </div>
  );
}
```

**Assumptions I made**
- Page size is 25, capped at 100. Adjust `DEFAULT_PAGE_SIZE` and `MAX_PAGE_SIZE` if you want different values.
- Page-based pagination is fine. For a very large table, offset pagination gets slower on deep pages, and keyset pagination on `(created_at, id)` would be better.
- Out-of-range pages return 404. The frontend's disabled buttons prevent this in normal use.

**Verify**
- `curl -s "http://localhost:8000/api/reports/?page=2&page_size=25" | jq '.page, .num_pages, .count'` should return page 2 with the expected totals.
- Run the endpoint in a Django test with `django_assert_max_num_queries(3)` to confirm the N+1 is gone.
- Run `EXPLAIN ANALYZE` on the generated SQL (`str(qs[:25].query)`) to check that the index is used.
- Load `/reports` in the browser and confirm Next/Prev move through pages with no duplicate or missing rows.

**Watch for**
- `Paginator.count` runs a `COUNT(*)` on every request. If the table is large and the count is still slow, cache it for a short time or drop the total from the response.
