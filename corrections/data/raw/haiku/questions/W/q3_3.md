## Approach

Merging is risky mostly because dropping a table can't be undone. So I'd split it into three migrations that ship separately:

1. **Additive:** add any fields the keeper is missing, then copy rows from the redundant table into it (data migration, idempotent).
2. **Repoint:** move foreign keys and code references from the old model to the keeper.
3. **Drop:** delete the old model only after step 2 has run in production and been checked.

Steps 1 and 2 are reversible. Step 3 is the only irreversible one, so it goes last and alone.

```python
# customers/migrations/0002_merge_customer_data.py
from django.db import migrations
from django.db.models import Model


def copy_legacy_customers(apps, schema_editor) -> None:
    Customer = apps.get_model("customers", "Customer")
    LegacyCustomer = apps.get_model("customers", "LegacyCustomer")

    existing = {c.email.lower(): c for c in Customer.objects.all()}
    to_create: list[Customer] = []
    to_update: list[Customer] = []

    for old in LegacyCustomer.objects.iterator():
        key = old.email.lower()
        if key in existing:
            new = existing[key]
            # Fill gaps only; never overwrite data already on the keeper.
            changed = False
            for field in ("phone", "company", "billing_address"):
                if not getattr(new, field) and getattr(old, field):
                    setattr(new, field, getattr(old, field))
                    changed = True
            if changed:
                to_update.append(new)
        else:
            to_create.append(Customer(
                email=old.email,
                phone=old.phone,
                company=old.company,
                billing_address=old.billing_address,
                created_at=old.created_at,
            ))

    Customer.objects.bulk_create(to_create, batch_size=500)
    Customer.objects.bulk_update(to_update, ["phone", "company", "billing_address"], batch_size=500)


class Migration(migrations.Migration):
    dependencies = [("customers", "0001_initial")]

    operations = [
        migrations.RunPython(copy_legacy_customers, migrations.RunPython.noop),
    ]
```

Things to check before using this:
- Fields are placeholders. Match them to your real models.
- If `email` isn't unique on the keeper, duplicates will slip through. Dedupe first or add a unique constraint in a separate step.
- `bulk_create` skips `save()` and signals. If `Customer` has either, use a loop with `get_or_create`.
