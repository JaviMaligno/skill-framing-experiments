The helper lowercases the title, strips accents, turns each run of non-alphanumeric characters into a single hyphen, and caps the length at a word boundary. It uses only the standard library, so it works anywhere, including Django. I've assumed it goes in `invoices/slugs.py`. Adjust the path to match the repo.

```python
# invoices/slugs.py
from __future__ import annotations

import re
import unicodedata

_NON_ALNUM = re.compile(r"[^a-z0-9]+")
_FALLBACK = "invoice"


def invoice_slug(title: str, max_length: int = 80) -> str:
    """Return a URL-safe slug for an invoice title.

    >>> invoice_slug("Invoice #2024-017: Café & Co. (March)")
    'invoice-2024-017-cafe-co-march'
    """
    if max_length < 1:
        raise ValueError("max_length must be at least 1")

    # Decompose accents ("é" -> "e" + combining mark), then drop non-ASCII bytes.
    ascii_text = (
        unicodedata.normalize("NFKD", title)
        .encode("ascii", "ignore")
        .decode("ascii")
        .lower()
    )
    slug = _NON_ALNUM.sub("-", ascii_text).strip("-")

    if len(slug) > max_length:
        cut = slug[:max_length]
        # Avoid splitting a word: back up to the last hyphen unless the cut is already on one.
        if slug[max_length] != "-":
            cut = cut.rpartition("-")[0] or cut
        slug = cut.rstrip("-")

    return slug or _FALLBACK
```

Tests:

```python
# invoices/tests/test_slugs.py
import pytest

from invoices.slugs import invoice_slug


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("Invoice #2024-017: Café & Co. (March)", "invoice-2024-017-cafe-co-march"),
        ("  Acme   Corp  ", "acme-corp"),
        ("ALL CAPS", "all-caps"),
        ("Déjà vu—Naïve", "deja-vu-naive"),
        ("!!!", "invoice"),
        ("", "invoice"),
    ],
)
def test_invoice_slug(title: str, expected: str) -> None:
    assert invoice_slug(title) == expected


def test_truncates_at_word_boundary() -> None:
    slug = invoice_slug("alpha beta gamma delta", max_length=12)
    assert slug == "alpha-beta"
    assert len(slug) <= 12


def test_truncation_never_leaves_trailing_hyphen() -> None:
    assert invoice_slug("alpha beta", max_length=6) == "alpha"


def test_rejects_zero_max_length() -> None:
    with pytest.raises(ValueError):
        invoice_slug("x", max_length=0)
```

To verify, run `python -m pytest invoices/tests/test_slugs.py -v` from the repo root. It should pass with no dependencies beyond pytest.

Two things to know:
- **Transliteration is basic.** Characters with no ASCII decomposition are dropped, so "Straße" becomes "strae". If that matters for your invoices, add a small replacement map (for example `"ß": "ss"`) before the NFKD step.
- **Django alternative.** `django.utils.text.slugify` already does the accent stripping and hyphenation. This helper adds the length cap and the fallback for empty results. If you don't need either, `slugify(title)` is enough.
