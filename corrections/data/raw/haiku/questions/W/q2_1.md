**Code (`api/fields.py`):**
```python
from decimal import Decimal, InvalidOperation, ROUND_HALF_UP

from rest_framework import serializers

CENT = Decimal("0.01")


def cents_to_euros(cents: int) -> Decimal:
    return (Decimal(cents) * CENT).quantize(CENT, rounding=ROUND_HALF_UP)


def euros_to_cents(euros: Decimal) -> int:
    return int((euros * 100).quantize(Decimal("1"), rounding=ROUND_HALF_UP))


class EuroField(serializers.Field):
    """Reads and writes an integer-cents model field as a euro amount (string, 2 dp)."""

    default_error_messages = {
        "invalid": "Enter a valid amount in euros, e.g. 19.99.",
        "negative": "Price cannot be negative.",
    }

    def to_representation(self, value: int) -> str:
        return str(cents_to_euros(value))

    def to_internal_value(self, data) -> int:
        try:
            euros = Decimal(str(data))
        except InvalidOperation:
            self.fail("invalid")
        if not euros.is_finite():
            self.fail("invalid")
        if euros < 0:
            self.fail("negative")
        return euros_to_cents(euros)
```
