**Goal:** turn an invoice title such as `"Invoice #INV-2026/001 – Café Déjà Vu"` into a URL-safe slug such as `invoice-inv-2026-001-cafe-deja-vu`.

**Approach:** one stdlib-only function, no new dependencies. It transliterates accented characters to ASCII, lowercases, collapses runs of non-alphanumerics into single hyphens, and truncates on a word boundary. Put it in whichever module holds your invoice code.

**Assumptions:**
- Slugs only need to be unique per invoice. Collision handling (such as appending the invoice ID) is left to the caller.
- An title with no usable characters raises `ValueError` rather than returning an empty slug, so a bad route can't be created silently.
- Default max length is 80 characters.

```python
from __future__ import annotations

import re
import unicodedata

_NON_ALNUM: re.Pattern[str] = re.compile(r"[^a-z0-9]+")


def slugify_invoice_title(title: str, max_length: int = 80) -> str:
    """Convert an invoice title into a lowercase, ASCII, hyphen-separated slug.

    Raises:
        ValueError: if max_length < 1 or the title yields no usable characters.
    """
    if max_length < 1:
        raise ValueError("max_length must be >= 1")

    # Decompose accents (é -> e + combining mark), then drop non-ASCII bytes.
    decomposed = unicodedata.normalize("NFKD", title)
    ascii_only = decomposed.encode("ascii", "ignore").decode("ascii")

    slug = _NON_ALNUM.sub("-", ascii_only.lower()).strip("-")
    if not slug:
        raise ValueError(f"Cannot build a slug from title: {title!r}")

    if len(slug) > max_length:
        # Cut on the last hyphen that fits, so words aren't split mid-way.
        head = slug[: max_length + 1]
        if "-" in head:
            slug = head[: head.rindex("-")]
        else:
            # One unbroken word longer than max_length: hard cut.
            slug = slug[:max_length]

    return slug
```

**Tests** (save as `tests/test_slugify_invoice_title.py`):

```python
import pytest

from your_module import slugify_invoice_title  # adjust the import


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("Invoice #INV-2026/001 – Café Déjà Vu", "invoice-inv-2026-001-cafe-deja-vu"),
        ("  Acme   Corp  ", "acme-corp"),
        ("Q3 — Consulting (draft)", "q3-consulting-draft"),
        ("ÉÈÀ", "ea"),  # NFKD keeps the base letters
    ],
)
def test_slugify_basic(title: str, expected: str) -> None:
    assert slugify_invoice_title(title) == expected


def test_truncates_on_word_boundary() -> None:
    slug = slugify_invoice_title("alpha beta gamma delta", max_length=12)
    assert slug == "alpha-beta"


def test_hard_cuts_single_long_word() -> None:
    assert slugify_invoice_title("x" * 20, max_length=8) == "x" * 8


def test_no_trailing_hyphen_after_truncation() -> None:
    slug = slugify_invoice_title("aaaa bbbb", max_length=5)
    assert not slug.endswith("-")


@pytest.mark.parametrize("title", ["", "   ", "!!!", "日本語"])
def test_empty_result_raises(title: str) -> None:
    with pytest.raises(ValueError):
        slugify_invoice_title(title)
```
