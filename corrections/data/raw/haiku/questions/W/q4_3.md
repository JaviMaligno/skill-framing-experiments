**1. New component: `src/components/CopyLinkButton.tsx`**

```tsx
import { useEffect, useState } from "react";

type CopyLinkButtonProps = {
  url?: string; // defaults to the current page URL
  label?: string;
};

export function CopyLinkButton({ url, label = "Copy link" }: CopyLinkButtonProps) {
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    if (!copied) return;
    const timer = window.setTimeout(() => setCopied(false), 2000);
    return () => window.clearTimeout(timer);
  }, [copied]);

  async function handleCopy() {
    const target = url ?? window.location.href;
    try {
      await navigator.clipboard.writeText(target);
      setCopied(true);
    } catch {
      // Clipboard blocked (insecure context or permission denied): let the user copy manually.
      window.prompt("Copy this link:", target);
    }
  }

  return (
    <button
      type="button"
      onClick={handleCopy}
      title="Copy link to this invoice"
      aria-live="polite"
    >
      {copied ? "Copied!" : label}
    </button>
  );
}
```
