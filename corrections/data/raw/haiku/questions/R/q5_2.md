**Request:** A Python helper that converts an invoice title into a lowercase, ASCII, hyphen-separated URL slug.

**Assumptions:** Accents are transliterated to ASCII (`Factura de España` → `factura-de-espana`), any run of non-alphanumeric characters becomes one hyphen, and slugs are capped at 80 characters. An empty result falls back to `invoice`.

**Approach:** One pure function in a new module, `invoice_slug.py`, with no dependencies beyond the standard library, plus a pytest file. No billing code is touched, so no flag is needed.

## `invoice_slug.py`

```python
from __future__ import annotations

import re
import unicodedata

_NON_ALNUM: re.Pattern[str] = re.compile(r"[^a-z0-9]+")


def slugify_invoice_title(title: str, max_length: int = 80) -> str:
    """Convert an invoice title into a URL-safe slug.

    >>> slugify_invoice_title("Invoice #2026/001 – Acme Ltd.")
    'invoice-2026-001-acme-ltd'
    >>> slugify_invoice_title("Factura de España")
    'factura-de-espana'
    >>> slugify_invoice_title("   ***   ")
    'invoice'
    """
    if max_length < 1:
        raise ValueError("max_length must be at least 1")

    # Decompose accented characters, then drop anything that is not ASCII.
    ascii_text: str = (
        unicodedata.normalize("NFKD", title)
        .encode("ascii", "ignore")
        .decode("ascii")
    )

    slug: str = _NON_ALNUM.sub("-", ascii_text.lower()).strip("-")
    slug = slug[:max_length].rstrip("-")

    return slug or "invoice"
```

## `test_invoice_slug.py`

```python
import pytest

from invoice_slug import slugify_invoice_title


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("Invoice #2026/001 – Acme Ltd.", "invoice-2026-001-acme-ltd"),
        ("Factura de España", "factura-de-espana"),
        ("  Leading and trailing  ", "leading-and-trailing"),
        ("Multiple---separators___here", "multiple-separators-here"),
        ("ALL CAPS", "all-caps"),
        ("   ***   ", "invoice"),
        ("", "invoice"),
        ("日本語", "invoice"),
    ],
)
def test_slugify(title: str, expected: str) -> None:
    assert slugify_invoice_title(title) == expected


def test_max_length_truncates_without_trailing_hyphen() -> None:
    slug = slugify_invoice_title("aaaa bbbb cccc", max_length=5)
    assert slug == "aaaa"
    assert not slug.endswith("-")


def test_max_length_must_be_positive() -> None:
    with pytest.raises(ValueError):
        slugify_invoice_title("x", max_length=0)
```

## Verify

From the directory containing both files:

```powershell
python -m doctest -v invoice_slug.py
python -m pytest test_invoice_slug.py -v
```

Both should pass. If you want a different transliteration (for example, `ß` → `ss` or keeping underscores), the only line to change is the `_NON_ALNUM` pattern or the normalization step.
