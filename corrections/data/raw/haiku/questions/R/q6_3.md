```python
from __future__ import annotations

from django.db.models import QuerySet
from django.views.generic import ListView

from reports.models import Report


class ReportListView(ListView):
    model = Report
    template_name = "reports/report_list.html"
    context_object_name = "reports"
    paginate_by = 25
    # Adjust to the FKs the list template actually renders.
    # Example: select_related("owner", "project")

    def get_queryset(self) -> QuerySet[Report]:
        # Stable ordering is required for pagination; add pk as a tiebreaker.
        return (
            Report.objects.all()
            .order_by("-created_at", "-pk")
        )
```

If you already have a function view, the equivalent is:

```python
from django.core.paginator import Paginator

def report_list(request):
    qs = Report.objects.order_by("-created_at", "-pk")
    page_obj = Paginator(qs, 25).get_page(request.GET.get("page"))
    return render(request, "reports/report_list.html", {"page_obj": page_obj, "reports": page_obj.object_list})
```

`get_page()` returns the last page for out-of-range numbers and page 1 for garbage input, so it never 404s.

```python
import pytest
from django.urls import reverse

from reports.models import Report


@pytest.mark.django_db
def test_report_list_is_paginated(client, django_user_model) -> None:
    Report.objects.bulk_create([Report(title=f"r{i}") for i in range(60)])
    url = reverse("reports:list")  # adjust the URL name

    resp = client.get(url)
    assert resp.status_code == 200
    assert len(resp.context["reports"]) == 25

    resp = client.get(url, {"page": 3})
    assert len(resp.context["reports"]) == 10

    resp = client.get(url, {"page": 999})  # out of range, falls back to last page
    assert resp.status_code == 200


@pytest.mark.django_db
def test_report_list_query_count(client, django_assert_max_num_queries) -> None:
```
