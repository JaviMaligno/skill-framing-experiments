**Request:** A helper that converts an invoice title (for example, "Factura Nº 2026/034 — Café ñandú") into a lowercase, ASCII-only, hyphen-separated URL slug.

**Assumptions:** Accents fold to ASCII (`é` → `e`), `º` becomes `o`, and characters with no ASCII decomposition, such as `ø` or `ß`, get a small transliteration map. Empty results fall back to `"invoice"`. Slugs are capped at 80 characters and truncated at a hyphen boundary.

**Approach:** One module with one public function, `invoice_slug`, and a pytest file next to it. No dependencies beyond the standard library.

**`invoice_slug.py`**

```python
from __future__ import annotations

import re
import unicodedata

_NON_ALNUM = re.compile(r"[^a-z0-9]+")
# Letters that NFKD does not decompose into ASCII.
_TRANSLITERATE = str.maketrans({
    "ß": "ss", "ø": "o", "Ø": "O", "æ": "ae", "Æ": "AE",
    "œ": "oe", "Œ": "OE", "đ": "d", "Đ": "D", "ł": "l", "Ł": "L",
})


def invoice_slug(title: str, max_length: int = 80, fallback: str = "invoice") -> str:
    """Turn an invoice title into a URL slug.

    >>> invoice_slug("Factura Nº 2026/034 — Café ñandú")
    'factura-no-2026-034-cafe-nandu'
    """
    if max_length < 1:
        raise ValueError("max_length must be at least 1")

    text = title.translate(_TRANSLITERATE)
    # Split accents off base letters ("é" -> "e" + combining mark), then drop non-ASCII.
    ascii_text = unicodedata.normalize("NFKD", text).encode("ascii", "ignore").decode("ascii")

    slug = _NON_ALNUM.sub("-", ascii_text.lower()).strip("-")
    if not slug:
        return fallback
    if len(slug) <= max_length:
        return slug

    # Cut at max_length, then back up to the last hyphen so no word is split.
    cut = slug[:max_length]
    if "-" in cut:
        cut = cut.rsplit("-", 1)[0]
    return cut.strip("-") or fallback
```

**`test_invoice_slug.py`**

```python
import pytest

from invoice_slug import invoice_slug


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("Factura Nº 2026/034 — Café ñandú", "factura-no-2026-034-cafe-nandu"),
        ("  ¡Invoice #42!  ", "invoice-42"),
        ("Søren's Bøx", "soren-s-box"),
        ("Straße 5", "strasse-5"),
        ("!!!", "invoice"),
        ("", "invoice"),
    ],
)
def test_slug(title: str, expected: str) -> None:
    assert invoice_slug(title) == expected


def test_truncates_at_word_boundary() -> None:
    slug = invoice_slug("word " * 30, max_length=20)
    assert len(slug) <= 20
    assert not slug.endswith("-")
    assert set(slug.split("-")) == {"word"}


def test_rejects_non_positive_max_length() -> None:
    with pytest.raises(ValueError):
        invoice_slug("x", max_length=0)
```
