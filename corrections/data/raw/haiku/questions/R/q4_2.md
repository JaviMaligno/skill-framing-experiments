**Approach**

3. Use `navigator.clipboard` with a fallback for non-secure contexts.

**1. New file: `src/components/CopyLinkButton.tsx`**

```tsx
import { useEffect, useState } from "react";

type Props = {
  url: string;
  label?: string;
};

export function CopyLinkButton({ url, label = "Copy link" }: Props) {
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    if (!copied) return;
    const t = setTimeout(() => setCopied(false), 2000);
    return () => clearTimeout(t);
  }, [copied]);

  async function handleCopy() {
    try {
      if (navigator.clipboard?.writeText) {
        await navigator.clipboard.writeText(url);
      } else {
        // Fallback for non-secure contexts (plain http, older browsers)
        const ta = document.createElement("textarea");
        ta.value = url;
        ta.setAttribute("readonly", "");
        ta.style.position = "fixed";
        ta.style.opacity = "0";
        document.body.appendChild(ta);
        ta.select();
        document.execCommand("copy");
        document.body.removeChild(ta);
      }
      setCopied(true);
    } catch {
      // Clipboard denied; leave the button unchanged so the user can retry
    }
  }

  return (
    <button
      type="button"
      onClick={handleCopy}
      aria-label={`${label} for this invoice`}
      className="ml-2 inline-flex items-center rounded border px-2 py-1 text-sm hover:bg-gray-50"
    >
      {copied ? "Copied" : label}
    </button>
  );
}
```
