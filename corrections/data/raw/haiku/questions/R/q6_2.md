**Request:** Add pagination to the `/reports` page so it loads a page of reports at a time instead of all of them.

**Assumptions** (tell me if any are wrong and I'll adjust):
- The page is a React component that fetches from a DRF endpoint, `GET /api/reports/`, backed by a `Report` model with `created_at` and `id`.
- Page size is 25, with `?page_size=` allowed up to 100.
- No billing code is involved, so no feature flag is needed. No migration is needed either.

**Approach**
1. Backend: add a DRF pagination class and a stable ordering on the queryset. Ordering must be deterministic or pages can repeat or skip rows. Add `select_related` for any foreign keys the serializer reads, since a slow list usually has N+1 queries too.
2. Frontend: request `?page=N`, read `count`/`next`/`previous`, and render Previous/Next controls.
3. Verify with a test and a browser check.

`reports/views.py`:

from rest_framework import viewsets

from .models import Report
from .pagination import ReportPagination
from .serializers import ReportSerializer


class ReportViewSet(viewsets.ReadOnlyModelViewSet):
    serializer_class = ReportSerializer
    pagination_class = ReportPagination

    def get_queryset(self):
        return (
            Report.objects.select_related("owner")  # add every FK the serializer reads
            .order_by("-created_at", "-id")
        )

**2. Frontend**

`src/pages/ReportsPage.tsx`, replacing the current list. I'm assuming a `Report` type with `id` and `name`, and plain `fetch`. If you use an API client helper for auth, swap it into the `fetch` call.
```tsx
import { useEffect, useState } from "react";

type Report = { id: number; name: string };
type Page<T> = { count: number; next: string | null; previous: string | null; results: T[] };

const PAGE_SIZE = 25;

export function ReportsPage() {
  const [page, setPage] = useState(1);
  const [data, setData] = useState<Page<Report> | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const controller = new AbortController();
    setLoading(true);
    fetch(`/api/reports/?page=${page}&page_size=${PAGE_SIZE}`, {
      signal: controller.signal,
      credentials: "include",
    })
      .then((r) => {
        if (!r.ok) throw new Error(`HTTP ${r.status}`);
        return r.json() as Promise<Page<Report>>;
      })
      .then((d) => {
        setData(d);
        setError(null);
      })
      .catch((e) => {
        if (e.name !== "AbortError") setError(String(e));
      })
      .finally(() => setLoading(false));
    return () => controller.abort();
  }, [page]);

  if (error) return <p>Could not load reports: {error}</p>;
  if (!data) return <p>Loading…</p>;

  const totalPages = Math.max(1, Math.ceil(data.count / PAGE_SIZE));

  return (
    <>
      <ul>
        {data.results.map((r) => (
          <li key={r.id}>{r.name}</li>
        ))}
      </ul>
      <nav>
        <button disabled={!data.previous || loading} onClick={() => setPage((p) => p - 1)}>
          Previous
        </button>
        <span>
          Page {page} of {totalPages}
        </span>
        <button disabled={!data.next || loading} onClick={() => setPage((p) => p + 1)}>
          Next
        </button>
      </nav>
    </>
  );
}
```

**3. Verify**

Backend test (`reports/tests/test_pagination.py`):
```python
from django.urls import reverse
from django.test.utils import CaptureQueriesContext
from django.db import connection
from rest_framework.test import APITestCase

from reports.models import Report


class ReportPaginationTests(APITestCase):
    def setUp(self):
        # create user/auth as your other API tests do
        Report.objects.bulk_create([Report(name=f"r{i}") for i in range(60)])

    def test_first_page_is_capped(self):
        res = self.client.get(reverse("report-list"))
        self.assertEqual(res.status_code, 200)
        self.assertEqual(res.data["count"], 60)
        self.assertEqual(len(res.data["results"]), 25)
        self.assertIsNotNone(res.data["next"])

    def test_last_page_has_remainder(self):
        res = self.client.get(reverse("report-list"), {"page": 3})
        self.assertEqual(len(res.data["results"]), 10)
        self.assertIsNone(res.data["next"])

    def test_query_count_does_not_scale_with_page(self):
        with CaptureQueriesContext(connection) as ctx:
            self.client.get(reverse("report-list"))
        self.assertLessEqual(len(ctx.captured_queries), 4)  # auth + count + page
```
