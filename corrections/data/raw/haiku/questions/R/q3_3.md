```python
from typing import Any

from django.apps.registry import Apps
from django.db import migrations
from django.db.backends.base.schema import BaseDatabaseSchemaEditor


def _shared_fields(src: Any, dst: Any) -> list[str]:
    dst_names = {f.name for f in dst._meta.concrete_fields if not f.primary_key}
    return [
        f.name
        for f in src._meta.concrete_fields
        if not f.primary_key and f.name in dst_names
    ]


def merge_forward(apps: Apps, schema_editor: BaseDatabaseSchemaEditor) -> None:
    Customer = apps.get_model("crm", "Customer")
    LegacyCustomer = apps.get_model("crm", "LegacyCustomer")

    fields = _shared_fields(LegacyCustomer, Customer)
    seen_emails = set(Customer.objects.values_list("email", flat=True))
    conflicts: list[str] = []
    batch: list[Any] = []

    for old in LegacyCustomer.objects.iterator(chunk_size=1000):
        if old.email in seen_emails:
            conflicts.append(old.email)
            continue
        seen_emails.add(old.email)
        batch.append(Customer(**{f: getattr(old, f) for f in fields}))
        if len(batch) >= 1000:
            Customer.objects.bulk_create(batch)
            batch.clear()

    if batch:
        Customer.objects.bulk_create(batch)

    if conflicts:
        # Logged, not raised, so the migration completes. Review these by hand.
        print(f"[merge_customers] {len(conflicts)} duplicate emails skipped")


class Migration(migrations.Migration):
    dependencies = [
        ("crm", "0010_previous"),  # replace with the latest crm migration
    ]

    operations = [
        migrations.RunPython(merge_forward, migrations.RunPython.noop),
    ]
```
