## Code to apply

```python
# pricing/money.py
from decimal import Decimal, ROUND_HALF_UP

CENTS_PER_EUR = Decimal(100)
CENT_Q = Decimal("0.01")


def cents_to_eur(cents: int) -> Decimal:
    return (Decimal(cents) / CENTS_PER_EUR).quantize(CENT_Q, rounding=ROUND_HALF_UP)


def eur_to_cents(eur: Decimal | str) -> int:
    return int((Decimal(eur) * CENTS_PER_EUR).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
```

```python
# api/v2/serializers.py  (v1 keeps the cents field unchanged)
from rest_framework import serializers
from pricing.money import cents_to_eur, eur_to_cents


class ProductV2Serializer(serializers.ModelSerializer):
    price = serializers.DecimalField(max_digits=10, decimal_places=2, min_value=0)

    class Meta:
        model = Product
        fields = ["id", "name", "price"]

    def to_representation(self, obj: Product) -> dict:
        data = super().to_representation(obj)
        data["price"] = str(cents_to_eur(obj.price_cents))
        return data

    def create(self, validated: dict) -> Product:
        validated["price_cents"] = eur_to_cents(validated.pop("price"))
        return super().create(validated)

    def update(self, instance: Product, validated: dict) -> Product:
        if "price" in validated:
            validated["price_cents"] = eur_to_cents(validated.pop("price"))
        return super().update(instance, validated)
```

def test_cents_to_eur_formats_two_places():

    assert cents_to_eur(5) == Decimal("0.05")


def test_eur_to_cents_round_trip():
    assert eur_to_cents("12.50") == 1250
    assert eur_to_cents(Decimal("0.99")) == 99
