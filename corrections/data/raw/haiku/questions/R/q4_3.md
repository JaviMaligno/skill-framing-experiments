**1. New component: `src/components/CopyLinkButton.tsx`**

```tsx
import { useEffect, useState } from "react";

type CopyLinkButtonProps = {
  url?: string;          // defaults to the current page URL
  resetAfterMs?: number; // how long "Copied" stays visible
};

export function CopyLinkButton({ url, resetAfterMs = 2000 }: CopyLinkButtonProps) {
  const [status, setStatus] = useState<"idle" | "copied" | "error">("idle");

  useEffect(() => {
    if (status === "idle") return;
    const timer = window.setTimeout(() => setStatus("idle"), resetAfterMs);
    return () => window.clearTimeout(timer);
  }, [status, resetAfterMs]);

  async function handleClick() {
    try {
      await navigator.clipboard.writeText(url ?? window.location.href);
      setStatus("copied");
    } catch {
      setStatus("error");
    }
  }

  const label =
    status === "copied" ? "Copied" : status === "error" ? "Copy failed" : "Copy link";

  return (
    <button
      type="button"
      onClick={handleClick}
      aria-label="Copy link to this invoice"
      aria-live="polite"
      className="rounded border px-2 py-1 text-sm hover:bg-gray-100"
    >
      {label}
    </button>
  );
}
```

**3. Test: `src/components/CopyLinkButton.test.tsx`**

```tsx
import { render, screen, fireEvent, waitFor } from "@testing-library/react";
import { describe, it, expect, vi, beforeEach } from "vitest";
import { CopyLinkButton } from "./CopyLinkButton";

describe("CopyLinkButton", () => {
  const writeText = vi.fn();

  beforeEach(() => {
    writeText.mockReset();
    Object.defineProperty(navigator, "clipboard", {
      value: { writeText },
      configurable: true,
    });
  });

  it("copies the current URL and shows confirmation", async () => {
    writeText.mockResolvedValue(undefined);
    render(<CopyLinkButton />);

    fireEvent.click(screen.getByRole("button", { name: /copy link/i }));

    await waitFor(() => expect(screen.getByText("Copied")).toBeInTheDocument());
    expect(writeText).toHaveBeenCalledWith(window.location.href);
  });

  it("shows an error when the clipboard is unavailable", async () => {
    writeText.mockRejectedValue(new Error("denied"));
    render(<CopyLinkButton />);

    fireEvent.click(screen.getByRole("button", { name: /copy link/i }));

    await waitFor(() => expect(screen.getByText("Copy failed")).toBeInTheDocument());
  });
});
```
