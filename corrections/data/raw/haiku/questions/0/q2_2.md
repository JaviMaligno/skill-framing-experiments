```python
# shared/money.py
from decimal import Decimal, ROUND_HALF_UP

def cents_to_eur(cents: int) -> str:
    """Convert integer cents to a EUR decimal string, e.g. 1999 -> '19.99'."""
    eur = (Decimal(cents) / Decimal(100)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    return str(eur)
```

```python

class PublicProductSerializer(serializers.ModelSerializer):

    price_eur = serializers.SerializerMethodField()

    def get_price_eur(self, obj) -> str:
        return cents_to_eur(obj.price)
```
