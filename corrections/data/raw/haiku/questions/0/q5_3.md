**Request:** A function that converts an invoice title such as `Invoice #2026/015 — Café Déjà Vu` into a URL-safe slug such as `invoice-2026-015-cafe-deja-vu`.

**Approach:** One new module, `invoice_slug.py`, with no dependencies beyond the standard library. It transliterates accents to ASCII, lowercases, collapses runs of other characters into single hyphens, and truncates at a word boundary. It raises `ValueError` when nothing usable remains, so an empty slug never reaches a URL. Tests go alongside it.

**`invoice_slug.py`**

```python
import re
import unicodedata

_NON_ALNUM = re.compile(r"[^a-z0-9]+")


def invoice_slug(title: str, max_length: int = 80) -> str:
    """Convert an invoice title into a URL-safe slug.

    >>> invoice_slug("Invoice #2026/015 — Café Déjà Vu")
    'invoice-2026-015-cafe-deja-vu'
    """
    if max_length < 1:
        raise ValueError("max_length must be at least 1")

    ascii_text = (
        unicodedata.normalize("NFKD", title)
        .encode("ascii", "ignore")
        .decode("ascii")
    )
    slug = _NON_ALNUM.sub("-", ascii_text.lower()).strip("-")

    if len(slug) > max_length:
        cut = slug[:max_length]
        # Cut at a hyphen when the limit lands inside a word.
        if slug[max_length] != "-":
            cut = cut.rsplit("-", 1)[0]
        slug = cut.rstrip("-")

    if not slug:
        raise ValueError(f"title {title!r} produces an empty slug")
    return slug
```

**`tests/test_invoice_slug.py`**

```python
import pytest

from invoice_slug import invoice_slug


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("Invoice #2026/015 — Café Déjà Vu", "invoice-2026-015-cafe-deja-vu"),
        ("  Acme   Corp  ", "acme-corp"),
        ("ACME_Q3-2026!!", "acme-q3-2026"),
        ("Déjà vu", "deja-vu"),
    ],
)
def test_slugifies(title: str, expected: str) -> None:
    assert invoice_slug(title) == expected


def test_truncates_at_word_boundary() -> None:
    assert invoice_slug("aaa bbb ccc", max_length=7) == "aaa-bbb"
    assert invoice_slug("aaa bbb ccc", max_length=6) == "aaa"


def test_no_trailing_hyphen_after_truncation() -> None:
    assert not invoice_slug("aaa bbb ccc", max_length=4).endswith("-")


@pytest.mark.parametrize("title", ["", "   ", "#!!", "—"])
def test_empty_slug_raises(title: str) -> None:
    with pytest.raises(ValueError):
        invoice_slug(title)


def test_rejects_non_positive_max_length() -> None:
    with pytest.raises(ValueError):
        invoice_slug("abc", max_length=0)
```

**Limitations:**
- Characters with no ASCII decomposition, such as `ß`, `ø`, and `æ`, are dropped rather than transliterated (`Straße` becomes `strae`). If your invoice titles include these, add a small mapping table before the NFKD step.
- Slugs are not checked for uniqueness. If two invoices can share a title, append the invoice number or ID to the slug.
