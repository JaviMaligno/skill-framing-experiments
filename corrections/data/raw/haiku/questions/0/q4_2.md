**1. New component: `src/components/CopyLinkButton.tsx`**

```tsx
import { useEffect, useRef, useState } from "react";

type Props = {
  /** URL to copy. Defaults to the current page URL. */
  url?: string;
  label?: string;
};

export function CopyLinkButton({ url, label = "Copy link" }: Props) {
  const [copied, setCopied] = useState(false);
  const timer = useRef<number | undefined>(undefined);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  async function handleCopy() {
    const target = url ?? window.location.href;
    try {
      // Requires a secure context (HTTPS or localhost), which Railway provides.
      await navigator.clipboard.writeText(target);
      setCopied(true);
      window.clearTimeout(timer.current);
      timer.current = window.setTimeout(() => setCopied(false), 2000);
    } catch {
      // Fallback when the Clipboard API is unavailable or denied.
      window.prompt("Copy this link:", target);
    }
  }

  return (
    <button
      type="button"
      onClick={handleCopy}
      aria-live="polite"
      title="Copy link to this invoice"
      className="rounded border px-2 py-1 text-sm hover:bg-gray-100"
    >
      {copied ? "Copied!" : label}
    </button>
  );
}
```
