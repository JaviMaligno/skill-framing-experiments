3. **Management command** (dry-run by default, deletes in batches, each batch in its own transaction):

   ```python
   # <app>/management/commands/prune_stale_users.py
   from datetime import timedelta
   from typing import Any

   from django.contrib.auth import get_user_model
   from django.core.management.base import BaseCommand, CommandParser
   from django.db import transaction
   from django.db.models import Q
   from django.utils import timezone

   User = get_user_model()


   class Command(BaseCommand):
       help = "Delete stale non-staff users. Dry run unless --execute is passed."

       def add_arguments(self, parser: CommandParser) -> None:
           parser.add_argument("--days", type=int, default=365)
           parser.add_argument("--batch-size", type=int, default=500)
           parser.add_argument("--execute", action="store_true")

       def handle(self, *args: Any, **opts: Any) -> None:
           cutoff = timezone.now() - timedelta(days=opts["days"])
           qs = User.objects.filter(is_staff=False, is_superuser=False).filter(
               Q(last_login__lt=cutoff)
               | Q(last_login__isnull=True, date_joined__lt=cutoff)
           )
           total = qs.count()
           self.stdout.write(f"{total} users match (cutoff {cutoff:%Y-%m-%d}).")
           if not opts["execute"]:
               self.stdout.write("Dry run. Re-run with --execute to delete.")
               return

           deleted = 0
           while True:
               ids = list(qs.values_list("pk", flat=True)[: opts["batch_size"]])
               if not ids:
                   break
               with transaction.atomic():
                   User.objects.filter(pk__in=ids).delete()
               deleted += len(ids)
               self.stdout.write(f"Deleted {deleted}/{total}")
           self.stdout.write(self.style.SUCCESS(f"Done. Deleted {deleted} users."))
   ```
