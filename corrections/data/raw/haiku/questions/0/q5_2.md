**Request:** Write a function that converts an invoice title into a URL-safe slug.

**Approach:** One module with a single pure function, `slugify_invoice_title`. It transliterates accented characters to ASCII, lowercases, collapses runs of non-alphanumeric characters into single hyphens, trims hyphens, truncates to a maximum length without leaving a trailing hyphen, and falls back to `invoice` if nothing usable remains.

**Code** (`invoices/slugs.py`):

```python
from __future__ import annotations

import re
import unicodedata

_NON_ALNUM: re.Pattern[str] = re.compile(r"[^a-z0-9]+")


def slugify_invoice_title(title: str, max_length: int = 80) -> str:
    """Return a URL-safe slug for an invoice title.

    >>> slugify_invoice_title("Invoice #2026-001 — Acme & Co.")
    'invoice-2026-001-acme-co'
    >>> slugify_invoice_title("Café Résumé")
    'cafe-resume'
    >>> slugify_invoice_title("!!!")
    'invoice'
    """
    if max_length < 1:
        raise ValueError("max_length must be at least 1")

    ascii_text: str = (
        unicodedata.normalize("NFKD", title)
        .encode("ascii", "ignore")
        .decode("ascii")
        .lower()
    )
    slug: str = _NON_ALNUM.sub("-", ascii_text).strip("-")

    if len(slug) > max_length:
        slug = slug[:max_length].rstrip("-")

    return slug or "invoice"
```

**Tests** (`invoices/test_slugs.py`):

```python
import pytest

from invoices.slugs import slugify_invoice_title


@pytest.mark.parametrize(
    ("title", "expected"),
    [
        ("Invoice #2026-001 — Acme & Co.", "invoice-2026-001-acme-co"),
        ("Café Résumé", "cafe-resume"),
        ("  spaces   and\ttabs  ", "spaces-and-tabs"),
        ("---weird---", "weird"),
        ("!!!", "invoice"),
        ("", "invoice"),
    ],
)
def test_slugify_basic(title: str, expected: str) -> None:
    assert slugify_invoice_title(title) == expected


def test_truncates_without_trailing_hyphen() -> None:
    slug = slugify_invoice_title("abc def ghi", max_length=4)
    assert slug == "abc"
    assert not slug.endswith("-")


def test_rejects_invalid_max_length() -> None:
    with pytest.raises(ValueError):
        slugify_invoice_title("x", max_length=0)
```

**Verify:**

```powershell
py -3.12 -m pytest invoices/test_slugs.py -q
py -3.12 -m doctest -v invoices/slugs.py
```

Both should pass. Note that characters with no ASCII decomposition, such as `ß` or CJK, are dropped, so a title made entirely of them falls back to `invoice`. If you need those transliterated, add a library such as `python-slugify` and keep this function as the fallback.
