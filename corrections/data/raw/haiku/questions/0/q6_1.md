**Approach**
1. `reports/views.py`: paginate the queryset, 25 per page, and fix the ordering so pages are stable.
2. `reports/templates/reports/report_list.html`: render the current page and previous/next links.
3. Remove N+1 queries on the rows (the likely cause of the slowness, alongside the missing LIMIT).
4. Optional migration for an index on the sort column, reviewed before it runs in production.
5. Tests for page boundaries and bad page numbers.

**1. View** (assumes a `Report` model with `created_at` and an `owner` FK)

```python
# reports/views.py
from __future__ import annotations

from django.core.paginator import EmptyPage, PageNotAnInteger, Paginator
from django.db.models import QuerySet
from django.http import HttpRequest, HttpResponse
from django.shortcuts import render

from .models import Report

REPORTS_PER_PAGE: int = 25


def report_list(request: HttpRequest) -> HttpResponse:
    queryset: QuerySet[Report] = (
        Report.objects.select_related("owner")   # add every FK the template touches
        .order_by("-created_at", "-pk")          # total order keeps pages stable
    )
    paginator = Paginator(queryset, REPORTS_PER_PAGE)
    raw_page: str = request.GET.get("page", "1")
    try:
        page_obj = paginator.page(raw_page)
    except PageNotAnInteger:
        page_obj = paginator.page(1)
    except EmptyPage:
        page_obj = paginator.page(paginator.num_pages)  # clamp to last page

    return render(request, "reports/report_list.html", {"page_obj": page_obj})
```

```html
<nav aria-label="Reports pages">

    <a href="?page={{ page_obj.previous_page_number }}">Previous</a>

  {% if page_obj.has_next %}

  {% endif %}
</nav>
```

**4. Tests**

```python
# reports/tests/test_report_list_pagination.py
from django.test import TestCase
from django.urls import reverse

from reports.models import Report
from reports.views import REPORTS_PER_PAGE


class ReportListPaginationTests(TestCase):
    def setUp(self) -> None:
        Report.objects.bulk_create(
            [Report(title=f"r{i}") for i in range(REPORTS_PER_PAGE + 5)]
        )

    def test_first_page_has_page_size_rows(self) -> None:
        response = self.client.get(reverse("report_list"))
        self.assertEqual(len(response.context["page_obj"]), REPORTS_PER_PAGE)
        self.assertTrue(response.context["page_obj"].has_next())

    def test_pages_do_not_overlap(self) -> None:
        first = self.client.get(reverse("report_list"), {"page": 1})
        second = self.client.get(reverse("report_list"), {"page": 2})
        ids_1 = {r.pk for r in first.context["page_obj"]}
        ids_2 = {r.pk for r in second.context["page_obj"]}
        self.assertTrue(ids_1.isdisjoint(ids_2))

    def test_bad_and_out_of_range_pages_do_not_500(self) -> None:
        for page in ("abc", "0", "9999"):
            response = self.client.get(reverse("report_list"), {"page": page})
            self.assertEqual(response.status_code, 200)

    def test_query_count_is_constant(self) -> None:
        with self.assertNumQueries(3):  # count + page select + session/auth; adjust to your baseline
            self.client.get(reverse("report_list"))
```

**Things to check**
