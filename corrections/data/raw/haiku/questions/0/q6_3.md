**1. `reports/views.py`**

```python
from typing import Any

from django.db.models import QuerySet
from django.views.generic import ListView

from .models import Report


class ReportListView(ListView):
    model = Report
    template_name = "reports/report_list.html"
    context_object_name = "reports"
    paginate_by = 25

    def get_queryset(self) -> QuerySet[Report]:
        # Stable ordering: without the pk tiebreaker, rows with equal
        # created_at can appear on two pages or on none.
        return (
            Report.objects.select_related("owner")  # FKs the list renders
            .defer("body")                          # large fields not shown in the list
            .order_by("-created_at", "-pk")
        )

    def get_context_data(self, **kwargs: Any) -> dict[str, Any]:
        ctx = super().get_context_data(**kwargs)
        paginator = ctx["paginator"]
        ctx["page_range"] = paginator.get_elided_page_range(
            ctx["page_obj"].number, on_each_side=2, on_ends=1
        )
        return ctx
```

import pytest
from django.db import connection
from django.test.utils import CaptureQueriesContext

from reports.models import Report


@pytest.mark.django_db
def test_report_list_is_paginated(client, django_user_model) -> None:
    user = django_user_model.objects.create_user("t@example.com", "pw")
    client.force_login(user)
    Report.objects.bulk_create([Report(title=f"r{i}") for i in range(60)])

    resp = client.get("/reports/")
    assert resp.status_code == 200
    assert len(resp.context["reports"]) == 25
    assert resp.context["page_obj"].paginator.num_pages == 3

    resp = client.get("/reports/?page=3")
    assert len(resp.context["reports"]) == 10


@pytest.mark.django_db
def test_report_list_query_count_is_constant(client, django_user_model) -> None:
    user = django_user_model.objects.create_user("t@example.com", "pw")
    client.force_login(user)
    Report.objects.bulk_create([Report(title=f"r{i}") for i in range(60)])

    with CaptureQueriesContext(connection) as ctx:
        client.get("/reports/")
    # Count + page select + session/user lookups; should not grow with page size.
    assert len(ctx.captured_queries) <= 5
