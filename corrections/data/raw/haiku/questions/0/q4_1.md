**Approach**
1. Add a reusable `CopyLinkButton` component. It uses `navigator.clipboard` with a `execCommand` fallback for non-secure contexts.
2. Render it next to the invoice number in the detail header.
3. No backend change. The link is the current page URL unless you pass one.

**1. New component** (`src/components/CopyLinkButton.tsx`)

```tsx
import { useEffect, useState } from "react";

type Props = {
  url?: string; // defaults to the current page URL
  className?: string;
};

export function CopyLinkButton({ url, className }: Props) {
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    if (!copied) return;
    const timer = setTimeout(() => setCopied(false), 2000);
    return () => clearTimeout(timer);
  }, [copied]);

  async function handleCopy() {
    const link = url ?? window.location.href;
    try {
      if (navigator.clipboard?.writeText) {
        await navigator.clipboard.writeText(link);
      } else {
        const ta = document.createElement("textarea");
        ta.value = link;
        ta.setAttribute("readonly", "");
        ta.style.position = "fixed";
        ta.style.opacity = "0";
        document.body.appendChild(ta);
        ta.select();
        document.execCommand("copy");
        document.body.removeChild(ta);
      }
      setCopied(true);
    } catch {
      // Clipboard blocked by the browser; leave the label unchanged.
    }
  }

  return (
    <button
      type="button"
      onClick={handleCopy}
      className={className}
      aria-live="polite"
    >
      {copied ? "Copied!" : "Copy link"}
    </button>
  );
}
```

**2. Place it next to the invoice number** (in the invoice detail page, wherever the header renders `invoice.number`)

```tsx

```
