```python
from django.db import migrations


def copy_legacy_customers(apps, schema_editor):
    Customer = apps.get_model("<app>", "Customer")
    LegacyCustomer = apps.get_model("<app>", "LegacyCustomer")

    existing_emails = set(Customer.objects.values_list("email", flat=True))
    to_create = []
    for old in LegacyCustomer.objects.iterator():
        if old.email in existing_emails:
            continue  # decide merge rule: keep newer, or merge fields
        to_create.append(Customer(email=old.email, name=old.name))
    Customer.objects.bulk_create(to_create, batch_size=1000)


class Migration(migrations.Migration):
    dependencies = [("<app>", "<previous_migration>")]

    operations = [
        migrations.RunPython(copy_legacy_customers, migrations.RunPython.noop),
    ]
```

Repoint foreign keys in a separate migration: add the new FK, backfill it from the old one, switch the code, and only then remove the old FK.
