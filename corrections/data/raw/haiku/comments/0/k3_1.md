```python
from typing import Any


def _check_payment(payment: Any) -> None:
    """Raise ValueError if the payment is not a dict with 'id' and 'ref' keys."""
    if not isinstance(payment, dict):
        raise ValueError(f"payment must be a dict, got {type(payment).__name__}")
    missing = {"id", "ref"} - payment.keys()
    if missing:
        raise ValueError(f"payment is missing keys: {sorted(missing)}")


def dedupe_payments(payments: list[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return one payment per ref, keeping the record with the lowest id."""
    canonical: dict[Any, dict[str, Any]] = {}
    for payment in payments:
        _check_payment(payment)
        ref = payment["ref"]
        current = canonical.get(ref)
        if current is None or payment["id"] < current["id"]:
            canonical[ref] = payment
    return list(canonical.values())
```

The result keeps refs in the order they first appear in the input, and each ref maps to its lowest-id record regardless of arrival order. Ids must be mutually comparable, so mixing ints and strings will raise a `TypeError`.
