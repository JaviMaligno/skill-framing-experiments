def format_invoice_number(year: int, seq: int) -> str:
    """Return the invoice number for a year and sequence, e.g. 'INV-2026-0042'."""
    if year < 1:
        raise ValueError(f"year must be positive, got {year}")
    if seq < 0:
        raise ValueError(f"seq must be non-negative, got {seq}")
    # Pad to four digits so invoice numbers sort correctly; above 9999 the number just grows.
    return f"INV-{year}-{seq:04d}"
