```python
import time
from collections.abc import Callable, Iterable
from typing import Any

# The provider drops requests that land inside its one-per-second window without
# an error, so 1.0 s leaves no margin for timer jitter.
MIN_INTERVAL_S: float = 1.1


def send_all(reminders: Iterable[Any], send: Callable[[Any], object]) -> None:
    """Return None after calling send on every reminder, at least 1.1 s apart."""
    if not callable(send):
        raise ValueError("send must be callable")
    items = list(reminders)
    for i, reminder in enumerate(items):
        if i:
            time.sleep(MIN_INTERVAL_S)
        send(reminder)
```

The sleep comes before each call except the first, so no delay follows the last send, and a slow `send` only increases the spacing.
