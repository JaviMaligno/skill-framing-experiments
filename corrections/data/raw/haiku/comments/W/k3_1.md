```python
from collections.abc import Iterable
from typing import Any


def dedupe_payments(payments: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return one payment per ref, keeping the lowest-id record for each ref."""
    canonical: dict[Any, dict[str, Any]] = {}
    for payment in payments:
        if not isinstance(payment, dict):
            raise ValueError(f"payment must be a dict, got {type(payment).__name__}")
        if "id" not in payment or "ref" not in payment:
            raise ValueError("payment must have 'id' and 'ref' keys")

        ref = payment["ref"]
        # Upstream order is arbitrary, so compare ids rather than keeping first or last.
        current = canonical.get(ref)
        if current is None or payment["id"] < current["id"]:
            canonical[ref] = payment

    # Replacing a value keeps the key's original slot, so refs stay in first-seen order.
    return list(canonical.values())
```

The function returns a list in first-seen ref order, and it raises `ValueError` for non-dict items or missing keys. Ids must be mutually comparable, so mixing ints and strings will raise `TypeError`.
