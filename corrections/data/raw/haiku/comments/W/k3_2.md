```python
from collections.abc import Iterable
from typing import Any


def dedupe_payments(payments: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    """Return one payment per ref, keeping the record with the lowest id."""
    canonical: dict[Any, dict[str, Any]] = {}
    for payment in payments:
        if not isinstance(payment, dict) or "id" not in payment or "ref" not in payment:
            raise ValueError(f"payment must be a dict with 'id' and 'ref': {payment!r}")
        ref = payment["ref"]
        current = canonical.get(ref)
        # Duplicates arrive out of order, so first-seen is not canonical; compare ids.
        if current is None or payment["id"] < current["id"]:
            canonical[ref] = payment
    return list(canonical.values())
```

Output order follows the first appearance of each ref, and ids must be mutually comparable (a mix of types raises TypeError).
